<FlowButtons onBack={() => setStep('card-details')} onNext={() => setStep('review')} nextLabel="Proceed to review" nextDisabled={!canProceedCvv} />
</StepContent>
)}

{step === 'review' && (
<StepContent title="Review your payment" subtitle="Confirm everything looks right">
<div className="review-card">
<ReviewRow icon={<Landmark size={15} />} label="Bank" value={form.bankName} />
<ReviewRow icon={<CreditCard size={15} />} label="Card" value={`•••• ${form.cardNumber.replace(/\s/g, '').slice(-4)}`} />
<ReviewRow icon={<User size={15} />} label="Cardholder" value={form.cardHolder.toUpperCase()} />
<ReviewRow icon={<Calendar size={15} />} label="Expiry" value={form.expiry} />
<ReviewRow icon={<Building2 size={15} />} label="Merchant" value={form.merchant || 'Card Payment'} />
<div className="review-amount">
<span>Amount</span><strong>₹ {Number(form.amount).toLocaleString('en-IN')}</strong>
</div>
</div>
<button className="primary-button" onClick={handleProcess}>
<LockKeyhole size={17} /> Pay securely ₹ {Number(form.amount).toLocaleString('en-IN')}

);
}

/* ── UPI FLOW ─────────────────────────────────────── */
function UpiFlow({ onBack, onToast, onDone }: { onBack: () => void; onToast: (m: string) => void; onDone: () => void }) {
const [step, setStep] = useState<UpiStep>('upi-id');
const [form, setForm] = useState<UpiFormData>({ upiId: '', pin: '', amount: '', merchant: '' });
const [result, setResult] = useState<{ status: 'approved' | 'blocked'; risk: RiskLevel; reason: string } | null>(null);
const [scannedQr, setScannedQr] = useState<ParsedUpiQr | null>(null);

const canProceedUpi = /^[a-zA-Z0-9.\-_]{2,}@[a-zA-Z]{2,}$/.test(form.upiId);
const canProceedAmount = form.amount.trim() !== '' && Number(form.amount) > 0;
const canProceedPin = form.pin.length >= 4;

const assessRisk = (): { status: 'approved' | 'blocked'; risk: RiskLevel; reason: string } => {
const amt = Number(form.amount);
const suspicious = ['fraud', 'scam', 'fake', 'unknown'];
const merchantLower = form.merchant.toLowerCase();
if (suspicious.some(s => merchantLower.includes(s))) return { status: 'blocked', risk: 'high', reason: 'Merchant flagged as suspicious. Transaction blocked for your safety.' };
if (amt > 25000) return { status: 'blocked', risk: 'high', reason: 'Amount exceeds UPI per-transaction limit of ₹25,000. Transaction blocked.' };
if (amt > 10000) return { status: 'approved', risk: 'medium', reason: 'High-value UPI transfer. Additional verification completed.' };
return { status: 'approved', risk: 'low', reason: 'UPI payment verified. Destination confirmed safe.' };
};

const submitTransaction = async (overrideStatus?: { status: 'approved' | 'blocked'; risk: RiskLevel; reason: string }) => {
const assessment = overrideStatus ?? assessRisk();
const txn: Omit<Transaction, 'id' | 'created_at'> = {
payment_type: 'upi',
bank_name: null,
card_number_last4: null,
card_holder: null,
upi_id: form.upiId,
amount: Number(form.amount),
merchant: form.merchant || 'UPI Payment',
risk_level: assessment.risk,
status: assessment.status,
risk_reason: assessment.reason,
};
const { error } = await supabase.from('transactions').insert(txn);
if (error) { onToast('Could not save transaction'); console.error(error); return; }
setResult(assessment);
setStep('result');
onDone();
};

const handleProcess = () => {
setStep('processing');
window.setTimeout(() => { submitTransaction(); }, 2200);
};

const handleQrScanned = (decodedText: string) => {
const parsed = parseUpiQr(decodedText);
if (!parsed) {
setScannedQr(null);
setStep('scan-failed');
return;
}
setScannedQr(parsed);
if (!parsed.valid) {
setStep('scan-failed');
return;
}

// Check if the scanned QR's UPI ID matches the user-entered UPI ID
if (form.upiId && parsed.upiId && form.upiId.toLowerCase() !== parsed.upiId.toLowerCase()) {
// UPI ID mismatch - this is a failed transaction
const mismatchResult = {
status: 'blocked' as const,
risk: 'high' as RiskLevel,
reason: `UPI ID mismatch: the QR code belongs to "${parsed.upiId}" but you entered "${form.upiId}". Transaction failed to protect you from sending money to the wrong recipient.`,
};
setForm(f => ({
...f,
merchant: parsed.merchantName || f.merchant,
amount: parsed.amount || f.amount,
}));
setStep('processing');
window.setTimeout(() => { submitTransaction(mismatchResult); }, 2200);
return;
}

// QR is valid and UPI ID matches (or no UPI ID was pre-entered)
setForm(f => ({
...f,
upiId: parsed.upiId,
merchant: parsed.merchantName || f.merchant,
amount: parsed.amount || f.amount,
}));
setStep('scan-result');
};

const handleQrError = (errorMsg: string) => {
console.error('QR scanner error:', errorMsg);
};

const reset = () => {
setForm({ upiId: '', pin: '', amount: '', merchant: '' });
setResult(null);
setScannedQr(null);
setStep('upi-id');
};

return (
<section className="flow-section">
<FlowHeader title="UPI / QR Payment" onBack={onBack} stepIndex={upiStepIndex(step)} steps={['UPI ID', 'Amount', 'PIN', 'Scan', 'Review', 'Done']} />
<div className="flow-card">
{step === 'upi-id' && (
<StepContent title="Enter your UPI ID" subtitle="Type your UPI ID or scan a QR code to pay">
<div className="form-field">
<label><Link2 size={14} /> UPI ID</label>
<input type="text" placeholder="yourname@bank" value={form.upiId} onChange={e => setForm({ ...form, upiId: e.target.value.toLowerCase().trim() })} className="form-input lower" />
{form.upiId && !canProceedUpi && <span className="field-hint">Enter a valid UPI ID (e.g. name@bank)</span>}
</div>
<div className="scan-divider"><span>or</span></div>
<button className="scan-launch-btn" onClick={() => setStep('scanning')}>
<ScanLine size={22} />
<span>Scan QR Code</span>
<small>Use camera or upload a photo</small>
</button>
<FlowButtons onNext={() => setStep('amount')} nextLabel="Continue" nextDisabled={!canProceedUpi} />
</StepContent>
)}

{step === 'amount' && (
<StepContent title="Enter payment amount" subtitle={`Paying to ${form.upiId}`}>
<div className="amount-display">
<span className="rupee">₹</span>
<input type="number" inputMode="decimal" placeholder="0" value={form.amount} onChange={e => setForm({ ...form, amount: e.target.value })} className="amount-input" />
</div>
<div className="form-field">
<label><Building2 size={14} /> Merchant / Note (optional)</label>
<input type="text" placeholder="e.g. FreshMart" value={form.merchant} onChange={e => setForm({ ...form, merchant: e.target.value })} className="form-input" />
</div>
<FlowButtons onBack={() => setStep('upi-id')} onNext={() => setStep('pin')} nextLabel="Continue" nextDisabled={!canProceedAmount} />
</StepContent>
)}

{step === 'pin' && (
<StepContent title="Enter UPI PIN" subtitle="Verify it's really you">
<div className="pin-dots">
{[0, 1, 2, 3].map(i => (
<span key={i} className={`pin-dot ${form.pin.length > i ? 'filled' : ''}`} />
))}
</div>
<input type="password" inputMode="numeric" placeholder="Enter 4-digit PIN" value={form.pin} onChange={e => setForm({ ...form, pin: e.target.value.replace(/\D/g, '').slice(0, 6) })} className="form-input pin-input" maxLength={6} autoFocus />
<div className="verify-notice">
<LockKeyhole size={18} />
<div><b>PIN encrypted</b><small>Your PIN is never stored or shared</small></div>
</div>
<FlowButtons onBack={() => setStep('amount')} onNext={() => setStep('scanning')} nextLabel="Scan QR to pay" nextDisabled={!canProceedPin} />
</StepContent>
)}

{step === 'scanning' && (
<StepContent title="Scan QR code" subtitle="Point your camera at a UPI QR or upload a photo">
<QrScanner onResult={handleQrScanned} onError={handleQrError} />
<div className="flow-buttons">
<button className="ghost-button" onClick={() => setStep(form.pin ? 'pin' : 'upi-id')}><ArrowLeft size={15} /> Back</button>
</div>
</StepContent>
)}

{step === 'scan-result' && scannedQr && (
<StepContent title="QR code verified" subtitle="Confirm the payment details from the QR code">
<div className="qr-result-card">
<div className="qr-result-icon"><Check size={32} /></div>
<div className="qr-result-info">
<div className="qr-info-row"><span>UPI ID</span><b>{scannedQr.upiId}</b></div>
{scannedQr.merchantName && <div className="qr-info-row"><span>Merchant</span><b>{scannedQr.merchantName}</b></div>}
{scannedQr.amount && <div className="qr-info-row"><span>Amount</span><b>₹ {Number(scannedQr.amount).toLocaleString('en-IN')}</b></div>}
{scannedQr.note && <div className="qr-info-row"><span>Note</span><b>{scannedQr.note}</b></div>}
</div>
</div>
<div className="scan-checks">
<ScanCheck label="Destination verified" done />
<ScanCheck label="Merchant safety checked" done />
<ScanCheck label="UPI ID match confirmed" done />
</div>
<button className="primary-button" onClick={() => setStep('review')}>
<Check size={17} /> Continue to review
</button>
<button className="ghost-button" onClick={() => setStep('scanning')}><RefreshCw size={15} /> Scan again</button>
</StepContent>
)}

{step === 'scan-failed' && (
<StepContent title="QR scan failed" subtitle="We couldn't read a valid UPI QR code">
<div className="scan-failed-card">
<div className="scan-failed-icon"><AlertTriangle size={40} /></div>
<p>This image doesn't contain a valid UPI payment QR code.</p>
<small>Make sure the QR is clearly visible and belongs to a UPI merchant.</small>
</div>
<button className="primary-button" onClick={() => setStep('scanning')}>
<RefreshCw size={17} /> Try scanning again
</button>
<button className="ghost-button" onClick={() => setStep('upi-id')}><ArrowLeft size={15} /> Enter UPI ID manually</button>
</StepContent>
)}

{step === 'review' && (
<StepContent title="Review your payment" subtitle="Confirm before sending">
<div className="review-card">
<ReviewRow icon={<Link2 size={15} />} label="UPI ID" value={form.upiId} />
<ReviewRow icon={<Building2 size={15} />} label="Merchant" value={form.merchant || 'UPI Payment'} />
<ReviewRow icon={<LockKeyhole size={15} />} label="PIN" value="••••" />
<div className="review-amount">
<span>Amount</span><strong>₹ {Number(form.amount).toLocaleString('en-IN')}</strong>
</div>
</div>
<button className="primary-button" onClick={handleProcess}>
<ShieldCheck size={17} /> Pay ₹ {Number(form.amount).toLocaleString('en-IN')}
</button>
<button className="ghost-button" onClick={() => setStep('scanning')}><ArrowLeft size={15} /> Back</button>
</StepContent>
)}

{step === 'processing' && <ProcessingScreen label="Processing UPI payment..." />}

{step === 'result' && result && (
<ResultScreen
status={result.status}
risk={result.risk}
reason={result.reason}
amount={Number(form.amount)}
merchant={form.merchant || 'UPI Payment'}
onNew={reset}
onHome={onBack}
/>
)}
</div>
</section>
);
}

const cardStepMap: Record<CardStep, number> = { bank: 0, 'card-details': 1, cvv: 2, review: 3, processing: 4, result: 4 };
const upiStepMap: Record<UpiStep, number> = { 'upi-id': 0, amount: 1, pin: 2, scanning: 3, 'scan-result': 3, 'scan-failed': 3, review: 4, processing: 5, result: 5 };
function cardStepIndex(s: CardStep): number { return cardStepMap[s] ?? 0; }
function upiStepIndex(s: UpiStep): number { return upiStepMap[s] ?? 0; }

/* ── SHARED FLOW COMPONENTS ───────────────────────── */
function FlowHeader({ title, onBack, stepIndex, steps }: { title: string; onBack: () => void; stepIndex: number; steps: string[] }) {
return (
<div className="flow-header">
<button className="back-button" onClick={onBack}><ArrowLeft size={18} /> Back to home</button>
<h2>{title}</h2>
{steps.length > 0 && (
<div className="step-progress">
{steps.map((s, i) => (
<div key={s} className={`step-pill ${i < stepIndex ? 'done' : ''} ${i === stepIndex ? 'current' : ''}`}>
<span className="step-num">{i + 1}</span><span className="step-label">{s}</span>
</div>
))}
</div>
)}
</div>
);
}

function StepContent({ title, subtitle, children }: { title: string; subtitle: string; children: React.ReactNode }) {
return (
<div className="step-content">
<div className="step-titles"><h3>{title}</h3><p>{subtitle}</p></div>
{children}
</div>
);
}

function FlowButtons({ onBack, onNext, nextLabel, nextDisabled }: { onBack?: () => void; onNext: () => void; nextLabel: string; nextDisabled: boolean }) {
return (
<div className="flow-buttons">
{onBack && <button className="ghost-button" onClick={onBack}><ArrowLeft size={15} /> Back</button>}

</div>
);
}

function ReviewRow({ icon, label, value }: { icon: React.ReactNode; label: string; value: string }) {

}

return (
<div className="processing-screen">
<div className="processing-spinner"><Loader size={40} className="spin" /></div>
<h3>{label}</h3>

</div>
);
}

}) {
return (
<div className="result-screen">
<div className={`result-icon ${status}`}>
{status === 'approved' ? <Check size={48} /> : <ShieldAlert size={48} />}
</div>
<h3>{status === 'approved' ? 'Payment Approved' : 'Payment Blocked'}</h3>

<p className="result-merchant">{merchant}</p>
<div className={`result-reason ${status}`}>
{risk === 'low' ? <ShieldCheck size={16} /> : risk === 'medium' ? <CircleAlert size={16} /> : <ShieldAlert size={16} />}
<span>{reason}</span>
</div>
<div className="result-buttons">
<button className="primary-button" onClick={onNew}>New payment <ArrowRight size={16} /></button>
<button className="ghost-button" onClick={onHome}>Back to home</button>
</div>
</div>
);
}
